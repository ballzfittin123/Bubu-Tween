local function new(class, props, parent)
    local inst = Instance.new(class)
    for k, v in pairs(props) do inst[k] = v end
    if parent then inst.Parent = parent end
    return inst
end

local function assetIdOf(str)
    if type(str) ~= "string" then return nil end
    local digits = str:match("%d+")
    return digits and tonumber(digits) or nil
end

local function fieldBounds(name)
    local a, b = FIELDS[name][1], FIELDS[name][2]
    return {
        minX = math.min(a.X, b.X), maxX = math.max(a.X, b.X),
        minZ = math.min(a.Y, b.Y), maxZ = math.max(a.Y, b.Y),
    }
end

--=============================================================
-- TOKEN DETECTION
--=============================================================
local function getTokenInfo(part)
    if not part or not part:IsA("BasePart") then return nil end

    local function check(id)
        if not id then return nil end
        if TOKEN_IDS[id]        then return { id = id, global = false } end
        if GLOBAL_TOKEN_IDS[id] then return { id = id, global = true  } end
        return nil
    end

    if part:IsA("MeshPart") then
        local r = check(assetIdOf(part.MeshId))
        if r then return r end
        r = check(assetIdOf(part.TextureID))
        if r then return r end
    end
    local decal = part:FindFirstChildOfClass("Decal")
    if decal then
        local r = check(assetIdOf(decal.Texture)); if r then return r end
    end
    local tex = part:FindFirstChildOfClass("Texture")
    if tex then
        local r = check(assetIdOf(tex.Texture)); if r then return r end
    end
    local sm = part:FindFirstChildOfClass("SpecialMesh")
    if sm then
        local r = check(assetIdOf(sm.MeshId)); if r then return r end
        r = check(assetIdOf(sm.TextureId)); if r then return r end
    end
    return nil
end

--=============================================================
-- PERSISTENCE
--=============================================================
local function canSave()
    return type(writefile) == "function" and type(readfile) == "function"
end
local function fileExists()
    if type(isfile) == "function" then return isfile(CONFIG_PATH) end
    return false
end

local function loadConfig()
    if not canSave() or not fileExists() then return end
    local ok, content = pcall(readfile, CONFIG_PATH)
    if not ok or type(content) ~= "string" then return end
    local ok2, data = pcall(function() return HttpService:JSONDecode(content) end)
    if not ok2 or type(data) ~= "table" then return end

    if type(data.enabled) == "boolean" then state.running = data.enabled end
    if type(data.tweenTime) == "number" then
        state.tweenTime = math.clamp(data.tweenTime, MIN_TWEEN, MAX_TWEEN)
    end
    if type(data.onlyInField) == "boolean" then
        state.onlyInField = data.onlyInField
    end
    if type(data.fields) == "table" then
        for name, val in pairs(data.fields) do
            if FIELDS[name] and type(val) == "boolean" then
                selectedFields[name] = val
            end
        end
    end
    if type(data.tokens) == "table" then
        for idStr, val in pairs(data.tokens) do
            local id = tonumber(idStr)
            if id and TOKEN_IDS[id] and type(val) == "boolean" then
                selectedTokens[id] = val
            end
        end
    end
    if type(data.globals) == "table" then
        for idStr, val in pairs(data.globals) do
            local id = tonumber(idStr)
            if id and GLOBAL_TOKEN_IDS[id] and type(val) == "boolean" then
                selectedGlobals[id] = val
            end
        end
    end
end

local function saveConfig()
    if not canSave() then return end
    local tokenMap, globalMap = {}, {}
    for id, val in pairs(selectedTokens) do tokenMap[tostring(id)] = val end
    for id, val in pairs(selectedGlobals) do globalMap[tostring(id)] = val end
    local data = {
        enabled     = state.running,
        tweenTime   = state.tweenTime,
        onlyInField = state.onlyInField,
        fields      = selectedFields,
        tokens      = tokenMap,
        globals     = globalMap,
    }
    local ok, encoded = pcall(function() return HttpService:JSONEncode(data) end)
    if not ok then return end
    pcall(writefile, CONFIG_PATH, encoded)
end

loadConfig()

local function refreshSlider()
    local alpha = (state.tweenTime - MIN_TWEEN) / (MAX_TWEEN - MIN_TWEEN)
    fill.Size = UDim2.new(alpha, 0, 1, 0)
    knob.Position = UDim2.new(alpha, 0, 0.5, 0)
    tweenLabel.Text = string.format("Tween Time: %.2f", state.tweenTime)

local sliding = false
local function applySliderInput(input)
    local rel = (input.Position.X - sliderArea.AbsolutePosition.X)
                / math.max(sliderArea.AbsoluteSize.X, 1)
    local alpha = math.clamp(rel, 0, 1)
    local t = MIN_TWEEN + alpha * (MAX_TWEEN - MIN_TWEEN)
    state.tweenTime = math.floor(t * 100 + 0.5) / 100
    refreshSlider()
end

sliderArea.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
    or input.UserInputType == Enum.UserInputType.Touch then
        sliding = true
        applySliderInput(input)
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End and sliding then
                sliding = false
                saveConfig()
            end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement
    or input.UserInputType == Enum.UserInputType.Touch) then
        applySliderInput(input)
    end
end)

local function makeCheckboxRow(parent, yPos, labelText, getSelected, setSelected)

        Size = UDim2.new(1, 0, 0, 24),
        Position = UDim2.fromOffset(0, yPos),
        BackgroundTransparency = 1,
        BackgroundColor3 = Color3.fromRGB(150, 80, 255),

    }, parent)

    local checkBox = new("Frame", {
        Size = UDim2.fromOffset(14, 14),
        Position = UDim2.fromOffset(10, 5),
        BackgroundColor3 = Color3.fromRGB(55, 45, 75),
        BorderSizePixel = 0, ZIndex = 7,
    }, item)
    new("UICorner", { CornerRadius = UDim.new(0, 3) }, checkBox)
    new("UIStroke", { Color = Color3.fromRGB(120, 60, 200), Thickness = 1 }, checkBox)

    local checkMark = new("TextLabel", {
        Size = UDim2.fromScale(1, 1),
        BackgroundTransparency = 1, Text = "✓",
        TextColor3 = Color3.fromRGB(255, 255, 255),
        TextSize = 11, Font = Enum.Font.GothamBold,
        ZIndex = 8, Visible = false,
    }, checkBox)

    new("TextLabel", {
        Size = UDim2.new(1, -32, 1, 0),
        Position = UDim2.fromOffset(32, 0),
        BackgroundTransparency = 1, Text = labelText,
        TextColor3 = Color3.fromRGB(225, 215, 245),
        TextSize = 13, Font = Enum.Font.Gotham,
        TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 7,
    }, item)

    local function refresh()
        if getSelected() then
            checkBox.BackgroundColor3 = Color3.fromRGB(150, 80, 255)
            checkMark.Visible = true
        else
            checkBox.BackgroundColor3 = Color3.fromRGB(55, 45, 75)
            checkMark.Visible = false
        end
    end
    refresh()

    item.MouseButton1Click:Connect(function()
        setSelected(not getSelected())
        refresh()
        priorityStale = true
        saveConfig()
    end)
    item.MouseEnter:Connect(function() item.BackgroundTransparency = 0.85 end)
    item.MouseLeave:Connect(function() item.BackgroundTransparency = 1 end)
end

    miniPressPos = UserInputService:GetMouseLocation()
end)

miniBtn.MouseButton1Click:Connect(function()
    if miniPressPos then
        local delta = (UserInputService:GetMouseLocation() - miniPressPos).Magnitude
        miniPressPos = nil
        if delta >= DRAG_THRESHOLD then
            return
        end
    end
    setMinimized(false)
end)

local function makeDraggable(handle, target)
    local dragging, dragStart, startPos
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos  = target.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            target.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
end
makeDraggable(titleBar, main)
makeDraggable(miniBtn, miniBtn)

local function refreshSwitch()

        switchBtn.Text = "ON"
        switchBtn.BackgroundColor3 = Color3.fromRGB(90, 200, 120)
    else
        switchBtn.Text = "OFF"
        switchBtn.BackgroundColor3 = Color3.fromRGB(60, 50, 80)
    end
end
refreshSwitch()

switchBtn.MouseButton1Click:Connect(function()

    if not state.running and currentTween then
        pcall(function() currentTween:Cancel() end)
        currentTween = nil
    end

    saveConfig()
end)

local scanRoots = {}
for _, name in ipairs(SEARCH_FOLDERS) do
    local f = workspace:FindFirstChild(name)
    if f then table.insert(scanRoots, f) end
end
if #scanRoots == 0 then scanRoots = { workspace } end

--=============================================================
-- FIELD HELPERS
--=============================================================
local function isInAnySelectedField(pos)
    for name, enabled in pairs(selectedFields) do
        if enabled then
            local b = fieldBounds(name)
            if pos.X >= b.minX and pos.X <= b.maxX
            and pos.Z >= b.minZ and pos.Z <= b.maxZ then
                return true
            end
        end
    end
    return false
end

-- Returns the name of the first selected field containing pos, or nil.
local function getFieldAt(pos)
    for name, enabled in pairs(selectedFields) do
        if enabled then
            local b = fieldBounds(name)
            if pos.X >= b.minX and pos.X <= b.maxX
            and pos.Z >= b.minZ and pos.Z <= b.maxZ then
                return name
            end
        end
    end
    return nil
end

--=============================================================
-- ELIGIBILITY
--  playerOrigin = current player HRP position (or nil).
--  When onlyInField is ON, a field token is only eligible when
--  the player is standing inside the SAME field as the token.
--  Global tokens are never affected.
--=============================================================
local function isEligible(part, info, playerOrigin)
    if not info then return false end
    if info.global then
        return selectedGlobals[info.id] == true
    end
    if selectedTokens[info.id] ~= true then
        return false
    end

    local pos = part.Position
    if not isInAnySelectedField(pos) then
        return false
    end

    if state.onlyInField then
        if not playerOrigin then return false end
        local playerField = getFieldAt(playerOrigin)
        if not playerField then return false end
        local tokenField = getFieldAt(pos)
        if tokenField ~= playerField then return false end
    end

    return true
end

--=============================================================
-- PRIORITY LIST
--=============================================================
local SORT_ORIGIN_TOLERANCE = 32

local function buildPriorityList(origin)
    local list = {}
    for part, info in pairs(activeTokens) do
        if part.Parent
        and not skipTokens[part]
        and isEligible(part, info, origin) then
            list[#list + 1] = {
                part = part,
                dist = (part.Position - origin).Magnitude,
            }
        end
    end
    table.sort(list, function(a, b) return a.dist < b.dist end)
    priorityList   = list
    priorityOrigin = origin
    priorityStale  = false
end

local function insertIntoPriorityList(part, info, origin)
    if not origin then return end
    if not isEligible(part, info, origin) then return end

    local entry = { part = part, dist = (part.Position - origin).Magnitude }
    local lo, hi = 1, #priorityList + 1
    while lo < hi do
        local mid = math.floor((lo + hi) / 2)
        if priorityList[mid].dist < entry.dist then
            lo = mid + 1
        else
            hi = mid
        end
    end
    table.insert(priorityList, lo, entry)
end

local function prunePriorityList(playerOrigin)
    while #priorityList > 0 do
        local entry = priorityList[1]
        local part = entry.part
        local info = activeTokens[part]
        if not part.Parent
        or not info
        or skipTokens[part]
        or not isEligible(part, info, playerOrigin) then
            table.remove(priorityList, 1)
        else
            break
        end
    end
end

--=============================================================
-- TOKEN TRACKING
--=============================================================
local function snapshotTargets()
    local list = {}
    for _, root in ipairs(scanRoots) do
        if root.Parent then
            for _, inst in ipairs(root:GetDescendants()) do
                if getTokenInfo(inst) then
                    list[#list + 1] = inst
                end
            end
        end
    end
    return list
end

local function trackPart(part)
    if not part or not part:IsA("BasePart") then return end
    if skipTokens[part] or activeTokens[part] then return end

    local info = getTokenInfo(part)
    if not info then return end

    activeTokens[part] = info

    if priorityOrigin then
        insertIntoPriorityList(part, info, priorityOrigin)
    else
        priorityStale = true
    end

    part.AncestryChanged:Connect(function(_, parent)
        if not parent then
            activeTokens[part] = nil
        end
    end)
end

local function considerDescendant(desc)
    trackPart(desc)
    local parent = desc.Parent
    if parent and parent:IsA("BasePart") then
        trackPart(parent)
    end
end

--=============================================================
-- INITIAL EXCLUSION + EVENT HOOKS
--=============================================================
task.spawn(function()
    if not game:IsLoaded() then game.Loaded:Wait() end
    task.wait(INITIAL_SCAN_DELAY)

    for _, root in ipairs(scanRoots) do
        root.DescendantAdded:Connect(considerDescendant)
    end

    local initial = snapshotTargets()
    for _, part in ipairs(initial) do
        skipTokens[part] = true
        activeTokens[part] = nil
    end
    exclusionsReady = true
    priorityStale = true

    warn(("[Jelly's Legit Script] Ready. Excluded %d initial token(s); roots: %s")
        :format(#initial, table.concat(
            (function()
                local names = {}
                for _, r in ipairs(scanRoots) do names[#names + 1] = r.Name end
                return names
            end)(), ", ")))
end)

--=============================================================
-- PICK NEXT TARGET
--  Rebuilds when the list is stale OR when:
--   - the player moved past SORT_ORIGIN_TOLERANCE, OR
--   - onlyInField is ON and the player entered a different field
--     (since eligibility may have changed for every token).
--=============================================================
local function pickTarget(origin)
    local needRebuild = priorityStale
        or not priorityOrigin
        or (origin - priorityOrigin).Magnitude > SORT_ORIGIN_TOLERANCE

    if not needRebuild and state.onlyInField then
        local curField     = getFieldAt(origin)
        local originField  = getFieldAt(priorityOrigin)
        if curField ~= originField then
            needRebuild = true
        end
    end

    if needRebuild then
        buildPriorityList(origin)
    end

    prunePriorityList(origin)

    local entry = priorityList[1]
    return entry and entry.part or nil
end

--=============================================================
-- MAIN LOOP
--=============================================================
task.spawn(function()
    while true do
        if state.running and exclusionsReady then
            local char = LocalPlayer.Character
            local hrp  = char and char:FindFirstChild("HumanoidRootPart")
            local hum  = char and char:FindFirstChildOfClass("Humanoid")

            if hrp and hum and hum.Health > 0 then
                local target = pickTarget(hrp.Position)

                if target and target.Parent then
                    skipTokens[target] = true
                    activeTokens[target] = nil
                    if #priorityList > 0 and priorityList[1].part == target then
                        table.remove(priorityList, 1)
                    end

                    local dist = (target.Position - hrp.Position).Magnitude
                    if dist > 0.5 then
                        local info = TweenInfo.new(
                            state.tweenTime,
                            Enum.EasingStyle.Linear,
                            Enum.EasingDirection.Out
                        )
                        local tween = TweenService:Create(hrp, info, {
                            CFrame = CFrame.new(target.Position)
                        })
                        currentTween = tween
                        tween:Play()
                        tween.Completed:Wait()
                        if currentTween == tween then
                            currentTween = nil
                        end
                    end
                    task.wait(0.05)
                else
                    task.wait(POLL_INTERVAL)
                end
            else
                task.wait(0.2)
            end
        else
            task.wait(0.1)
        end
    end
end)
