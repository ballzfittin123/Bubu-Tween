--=============================================================
-- UTILITIES
--=============================================================
local function new(class, props, parent)
    local inst = Instance.new(class)
    for k, v in pairs(props) do inst[k] = v end
    if parent then inst.Parent = parent end
    return inst
end

local function assetIdOf(str)
    if type(str) ~= "string" then return nil end
    local digits = str:match("%d+")
    return digits and tonumber(digits) or nil
end

local function fieldBounds(name)
    local a, b = FIELDS[name][1], FIELDS[name][2]
    return {
        minX = math.min(a.X, b.X), maxX = math.max(a.X, b.X),
        minZ = math.min(a.Y, b.Y), maxZ = math.max(a.Y, b.Y),
    }
end

--=============================================================
-- TOKEN DETECTION  (returns matching tokenId, or nil)
--=============================================================
local function getTokenId(part)
    if not part or not part:IsA("BasePart") then return nil end

    if part:IsA("MeshPart") then
        local mid = assetIdOf(part.MeshId)
        if mid and TOKEN_IDS[mid] then return mid end
        local tid = assetIdOf(part.TextureID)
        if tid and TOKEN_IDS[tid] then return tid end
    end

    local decal = part:FindFirstChildOfClass("Decal")
    if decal then
        local id = assetIdOf(decal.Texture)
        if id and TOKEN_IDS[id] then return id end
    end

    local tex = part:FindFirstChildOfClass("Texture")
    if tex then
        local id = assetIdOf(tex.Texture)
        if id and TOKEN_IDS[id] then return id end
    end

    local sm = part:FindFirstChildOfClass("SpecialMesh")
    if sm then
        local mid = assetIdOf(sm.MeshId)
        if mid and TOKEN_IDS[mid] then return mid end
        local tid = assetIdOf(sm.TextureId)
        if tid and TOKEN_IDS[tid] then return tid end
    end

    return nil
end

local function refreshSlider()
    local alpha = (state.tweenTime - MIN_TWEEN) / (MAX_TWEEN - MIN_TWEEN)
    fill.Size = UDim2.new(alpha, 0, 1, 0)
    knob.Position = UDim2.new(alpha, 0, 0.5, 0)
    tweenLabel.Text = string.format("Tween Time: %.2f", state.tweenTime)

local sliding = false
local function applySliderInput(input)
    local rel = (input.Position.X - sliderArea.AbsolutePosition.X)
                / math.max(sliderArea.AbsoluteSize.X, 1)
    local alpha = math.clamp(rel, 0, 1)
    local t = MIN_TWEEN + alpha * (MAX_TWEEN - MIN_TWEEN)
    state.tweenTime = math.floor(t * 100 + 0.5) / 100
    refreshSlider()
end

end)

--=============================================================
-- DRAG
--=============================================================
local function makeDraggable(handle, target)
    local dragging, dragStart, startPos
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos  = target.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            target.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
end
makeDraggable(titleBar, main)
makeDraggable(miniBtn, miniBtn)

--=============================================================
-- SCAN ROOTS
--=============================================================
local scanRoots = {}
for _, name in ipairs(SEARCH_FOLDERS) do
    local f = workspace:FindFirstChild(name)
    if f then table.insert(scanRoots, f) end
end
if #scanRoots == 0 then scanRoots = { workspace } end

--=============================================================
-- TOKEN TRACKING
--=============================================================
local function snapshotTargets()
    local list = {}
    for _, root in ipairs(scanRoots) do
        if root.Parent then
            for _, inst in ipairs(root:GetDescendants()) do
                if getTokenId(inst) then
                    list[#list + 1] = inst
                end
            end
        end
    end
    return list
end

local function trackPart(part)
    if not part or not part:IsA("BasePart") then return end
    if skipTokens[part] then return end
    if activeTokens[part] then return end

    local tokenId = getTokenId(part)
    if not tokenId then return end

    activeTokens[part] = tokenId

    part.AncestryChanged:Connect(function(_, parent)
        if not parent then
            activeTokens[part] = nil
        end
    end)
end

local function considerDescendant(desc)
    trackPart(desc)
    local parent = desc.Parent
    if parent and parent:IsA("BasePart") then
        trackPart(parent)
    end
end

--=============================================================
-- INITIAL EXCLUSION + EVENT HOOKS
--=============================================================
task.spawn(function()
    if not game:IsLoaded() then game.Loaded:Wait() end
    task.wait(INITIAL_SCAN_DELAY)

    -- Hook first so nothing spawned during the snapshot is missed.
    for _, root in ipairs(scanRoots) do
        root.DescendantAdded:Connect(considerDescendant)
    end

    local initial = snapshotTargets()
    for _, part in ipairs(initial) do
        skipTokens[part] = true
        activeTokens[part] = nil
    end
    exclusionsReady = true

    warn(("[CommonTokenTweener] Excluded %d initial token(s); roots: %s")
        :format(#initial, table.concat(
            (function()
                local names = {}
                for _, r in ipairs(scanRoots) do names[#names + 1] = r.Name end
                return names
            end)(), ", ")))
end)

--=============================================================
-- NEAREST TOKEN  (filters by selection + bounds)
--=============================================================
local function findNearest(fromPos, bounds)
    local keys = {}
    for part in pairs(activeTokens) do keys[#keys + 1] = part end

    local best, bestDist
    for _, part in ipairs(keys) do
        local tokenId = activeTokens[part]
        if tokenId
        and part.Parent
        and not skipTokens[part]
        and selectedTokens[tokenId] then
            local p = part.Position
            if p.X >= bounds.minX and p.X <= bounds.maxX
            and p.Z >= bounds.minZ and p.Z <= bounds.maxZ then
                local d = (p - fromPos).Magnitude
                if not bestDist or d < bestDist then
                    best, bestDist = part, d
                end
            end
        end
    end
    return best
end

--=============================================================
-- MAIN LOOP
--=============================================================
task.spawn(function()
    while true do
        if state.running and exclusionsReady then
            local char = LocalPlayer.Character
            local hrp  = char and char:FindFirstChild("HumanoidRootPart")
            local hum  = char and char:FindFirstChildOfClass("Humanoid")

            if hrp and hum and hum.Health > 0 then
                local target = findNearest(hrp.Position, fieldBounds(state.field))

                if target and target.Parent then
                    -- Consume BEFORE tweening: never re-select this instance.
                    skipTokens[target] = true
                    activeTokens[target] = nil

                    local dist = (target.Position - hrp.Position).Magnitude
                    if dist > 0.5 then
                        local info = TweenInfo.new(
                            state.tweenTime,
                            Enum.EasingStyle.Linear,
                            Enum.EasingDirection.Out
                        )
                        local tween = TweenService:Create(hrp, info, {
                            CFrame = CFrame.new(target.Position)
                        })
                        currentTween = tween
                        tween:Play()
                        tween.Completed:Wait()
                        if currentTween == tween then
                            currentTween = nil
                        end
                    end
                    task.wait(0.05)
                else
                    task.wait(POLL_INTERVAL)
                end
            else
                task.wait(0.2)
            end
        else
            task.wait(0.1)
        end
    end
end)
